## procedure 
```
--- IN 
CREATE PROCEDURE greet_user(IN username TEXT)
LANGUAGE plpgsql
AS $$
BEGIN
    RAISE NOTICE 'Bonjour, % !', username;
END;
$$;

CALL greet_user('Alice');

-- out

CREATE PROCEDURE get_pi(OUT pi_val NUMERIC)
LANGUAGE plpgsql
AS $$
BEGIN
    pi_val := 3.14159;
END;
$$;

-- Exécution
CALL get_pi(pi_val);

-- inout 
CREATE PROCEDURE double_value(INOUT num INT)
LANGUAGE plpgsql
AS $$
BEGIN
    num := num * 2;
END;
$$;

-- Exécution
CALL double_value(num := 5);
-- num vaut maintenant 10



--VARIADIC
CREATE PROCEDURE sum_numbers(VARIADIC nums INT[])
LANGUAGE plpgsql
AS $$
DECLARE
    total INT := 0;
BEGIN
    FOREACH n IN ARRAY nums
    LOOP
        total := total + n;
    END LOOP;
    RAISE NOTICE 'Total = %', total;
END;
$$;

-- Exécution
CALL sum_numbers(1, 2, 3, 4, 5);  -- affiche Total = 15
```
