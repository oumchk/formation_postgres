CREATE PROCEDURE sum_numbers(VARIADIC nums INT[])
LANGUAGE plpgsql
AS $$
DECLARE
    total INT := 0;
BEGIN
    FOREACH n IN ARRAY nums
    LOOP
        total := total + n;
    END LOOP;
    RAISE NOTICE 'Total = %', total;
END;
$$;

-- Exécution
CALL sum_numbers(1, 2, 3, 4, 5);  -- affiche Total = 15

CREATE OR REPLACE FUNCTION valider_salaire()
RETURNS TRIGGER
LANGUAGE plpgsql
AS $$
BEGIN
    -- Interdire la baisse de salaire
    IF TG_OP = 'UPDATE' AND NEW.sal < OLD.sal THEN
        RAISE EXCEPTION
            'Impossible de baisser le salaire de % à % pour %',
            OLD.sal, NEW.sal, OLD.ename;
    END IF;

    -- Interdire un salaire négatif ou nul
    IF NEW.sal <= 0 THEN
        RAISE EXCEPTION 'Le salaire doit être positif, reçu : %', NEW.salaire;
    END IF;

    IF NEW.sal <= 450 THEN
        RAISE EXCEPTION 'Le salaire doit être superieur au SMIG , reçu : %', NEW.salaire;
    END IF;

    IF NEW.sal > 10000 THEN
        RAISE EXCEPTION 'Le salaire ne peut être superieur au plafond, reçu : %', NEW.salaire;
    END IF;



    RETURN NEW;   -- tout est OK, on laisse passer
END;
$$;

CREATE TRIGGER trg_valider_salaire
    BEFORE INSERT OR UPDATE
    ON storeuser.emp
    FOR EACH ROW
    EXECUTE FUNCTION valider_salaire();

CREATE OR REPLACE PROCEDURE storeuser.transaction_emp()
LANGUAGE plpgsql
AS $$
DECLARE
    n               INT;
    i               INT := 1;
    v_sal_total     NUMERIC;
    v_sal_moyen     NUMERIC;
    v_sal_min       NUMERIC;
    v_sal_max       NUMERIC;
    v_emp           RECORD;
    v_next_empno    NUMERIC;
    v_erreur        BOOLEAN := FALSE;
BEGIN
    RAISE NOTICE '=== DEBUT DE LA TRANSACTION ===';
    RAISE NOTICE 'Table : storeuser.emp';
    RAISE NOTICE 'Heure : %', TO_CHAR(NOW(), 'DD/MM/YYYY HH24:MI:SS');


    RAISE NOTICE '';
    RAISE NOTICE '-- ETAPE 1 : Etat initial --';

    SELECT COUNT(*) INTO n FROM storeuser.emp;
    RAISE NOTICE 'Nombre employes    : %', n;

    SELECT
        SUM(sal),
        ROUND(AVG(sal), 2),
        MIN(sal),
        MAX(sal)
    INTO v_sal_total, v_sal_moyen, v_sal_min, v_sal_max
    FROM storeuser.emp;

    RAISE NOTICE 'Masse salariale    : %', v_sal_total;
    RAISE NOTICE 'Salaire moyen      : %', v_sal_moyen;
    RAISE NOTICE 'Salaire minimum    : %', v_sal_min;
    RAISE NOTICE 'Salaire maximum    : %', v_sal_max;

    SELECT MAX(empno) INTO v_next_empno FROM storeuser.emp;
    RAISE NOTICE 'MAX empno actuel   : %', v_next_empno;


    RAISE NOTICE '';
    RAISE NOTICE '-- ETAPE 2 : Liste employes par departement --';

    FOR v_emp IN
        SELECT
            e.empno,
            e.ename,
            e.job,
            e.sal,
            COALESCE(e.comm::VARCHAR, 'N/A') AS comm,
            d.dname,
            d.loc
        FROM storeuser.emp e
        JOIN storeuser.dept d ON d.deptno = e.deptno
        ORDER BY e.deptno, e.sal DESC
    LOOP
        RAISE NOTICE '  [%] % - % | sal=% | comm=% | dept=% (%)',
            v_emp.empno,
            v_emp.ename,
            v_emp.job,
            v_emp.sal,
            v_emp.comm,
            v_emp.dname,
            v_emp.loc;
    END LOOP;


    RAISE NOTICE '';
    RAISE NOTICE '-- ETAPE 3 : Augmentation salaires par JOB --';

    FOR v_emp IN
        SELECT empno, ename, job, sal
        FROM storeuser.emp
        ORDER BY empno
    LOOP
        DECLARE
            v_ancien  NUMERIC;
            v_nouveau NUMERIC;
            v_taux    NUMERIC;
            v_hausse  NUMERIC;
            v_pct     NUMERIC;
        BEGIN
            v_ancien := v_emp.sal;

            v_taux := CASE v_emp.job
                WHEN 'PRESIDENT' THEN 0.05
                WHEN 'MANAGER'   THEN 0.10
                WHEN 'ANALYST'   THEN 0.12
                WHEN 'CLERK'     THEN 0.15
                WHEN 'SALESMAN'  THEN 0.08
                ELSE                  0.05
            END;

            v_nouveau := ROUND(v_emp.sal * (1 + v_taux), 2);
            v_hausse  := ROUND(v_ancien * v_taux, 2);
            v_pct     := v_taux * 100;

            UPDATE storeuser.emp
            SET sal = v_nouveau
            WHERE empno = v_emp.empno;

            RAISE NOTICE '  % [%] : % -> % (+ % pct = + %)',
                v_emp.ename,
                v_emp.job,
                v_ancien,
                v_nouveau,
                v_pct,
                v_hausse;

        EXCEPTION
            WHEN check_violation THEN
                RAISE WARNING '  % : sal invalide -> ignore', v_emp.ename;
                v_erreur := TRUE;
        END;

        i := i + 1;
    END LOOP;


    RAISE NOTICE '';
    RAISE NOTICE '-- ETAPE 4 : Insertion nouveaux employes --';

    SELECT MAX(empno) INTO v_next_empno FROM storeuser.emp;

    BEGIN
        v_next_empno := v_next_empno + 100;
        INSERT INTO storeuser.emp (empno, ename, job, mgr, hiredate, sal, comm, deptno)
        VALUES (v_next_empno, 'MARTIN', 'CLERK', 7400, NOW(), 1100, NULL, 10);
        RAISE NOTICE '  OK empno=% MARTIN CLERK deptno=10', v_next_empno;

        v_next_empno := v_next_empno + 100;
        INSERT INTO storeuser.emp (empno, ename, job, mgr, hiredate, sal, comm, deptno)
        VALUES (v_next_empno, 'JOHNSON', 'ANALYST', 7400, NOW(), 2800, NULL, 20);
        RAISE NOTICE '  OK empno=% JOHNSON ANALYST deptno=20', v_next_empno;

        v_next_empno := v_next_empno + 100;
        INSERT INTO storeuser.emp (empno, ename, job, mgr, hiredate, sal, comm, deptno)
        VALUES (v_next_empno, 'ADAMS', 'SALESMAN', 7400, NOW(), 1600, 300, 30);
        RAISE NOTICE '  OK empno=% ADAMS SALESMAN deptno=30', v_next_empno;

    EXCEPTION
        WHEN foreign_key_violation THEN
            RAISE WARNING '  deptno inexistant -> annule';
            v_erreur := TRUE;
        WHEN unique_violation THEN
            RAISE WARNING '  empno deja existant -> annule';
            v_erreur := TRUE;
        WHEN check_violation THEN
            RAISE WARNING '  sal <= 0 refuse -> annule';
            v_erreur := TRUE;
    END;


    RAISE NOTICE '';
    RAISE NOTICE '-- ETAPE 5 : Attribution commissions SALESMAN --';

    UPDATE storeuser.emp
    SET comm = ROUND(sal * 0.20, 2)
    WHERE job  = 'SALESMAN'
      AND comm IS NULL;

    GET DIAGNOSTICS n = ROW_COUNT;
    RAISE NOTICE '  % SALESMAN mis a jour comm = 20 pct du sal', n;


    RAISE NOTICE '';
    RAISE NOTICE '-- ETAPE 6 : Bilan final --';

    SELECT COUNT(*) INTO n FROM storeuser.emp;
    RAISE NOTICE 'Total employes         : %', n;

    SELECT SUM(sal), ROUND(AVG(sal), 2)
    INTO v_sal_total, v_sal_moyen
    FROM storeuser.emp;

    RAISE NOTICE 'Nouvelle masse sal     : %', v_sal_total;
    RAISE NOTICE 'Nouveau salaire moyen  : %', v_sal_moyen;
    RAISE NOTICE 'Nouveau MAX empno      : %', v_next_empno;

    RAISE NOTICE '';
    RAISE NOTICE 'Resume par departement :';

    FOR v_emp IN
        SELECT
            d.dname,
            d.loc,
            COUNT(e.empno)              AS nb,
            COALESCE(SUM(e.sal), 0)     AS masse,
            COALESCE(ROUND(AVG(e.sal),2), 0) AS moy
        FROM storeuser.dept d
        LEFT JOIN storeuser.emp e ON e.deptno = d.deptno
        GROUP BY d.dname, d.loc
        ORDER BY d.dname
    LOOP
        RAISE NOTICE '  % (%) : % emp | masse=% | moy=%',
            v_emp.dname,
            v_emp.loc,
            v_emp.nb,
            v_emp.masse,
            v_emp.moy;
    END LOOP;


    RAISE NOTICE '';
    IF v_erreur THEN
        RAISE NOTICE 'Erreurs detectees -> ROLLBACK';
        ROLLBACK;
    ELSE
        RAISE NOTICE 'Aucune erreur -> COMMIT';
        COMMIT;
    END IF;

    RAISE NOTICE '=== FIN DE LA TRANSACTION ===';

END;
$$;

c.relname,
count(*) AS buffers
FROM pg_buffercache b
JOIN pg_class c
ON b.relfilenode = pg_relation_filenode(c.oid)
GROUP BY c.relname
ORDER BY buffers DESC;

SELECT n.nspname, c.relname, count(*) AS buffers
               FROM pg_buffercache b JOIN pg_class c
               ON b.relfilenode = pg_relation_filenode(c.oid) AND
                  b.reldatabase IN (0, (SELECT oid FROM pg_database
                                        WHERE datname = current_database()))
               JOIN pg_namespace n ON n.oid = c.relnamespace
               GROUP BY n.nspname, c.relname
               ORDER BY 3 DESC
               LIMIT 10;
