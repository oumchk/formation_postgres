CREATE PROCEDURE sum_numbers(VARIADIC nums INT[])
LANGUAGE plpgsql
AS $$
DECLARE
    total INT := 0;
BEGIN
    FOREACH n IN ARRAY nums
    LOOP
        total := total + n;
    END LOOP;
    RAISE NOTICE 'Total = %', total;
END;
$$;

-- Exécution
CALL sum_numbers(1, 2, 3, 4, 5);  -- affiche Total = 15

CREATE OR REPLACE FUNCTION valider_salaire()
RETURNS TRIGGER
LANGUAGE plpgsql
AS $$
BEGIN
    -- Interdire la baisse de salaire
    IF TG_OP = 'UPDATE' AND NEW.sal < OLD.sal THEN
        RAISE EXCEPTION
            'Impossible de baisser le salaire de % à % pour %',
            OLD.sal, NEW.sal, OLD.ename;
    END IF;

    -- Interdire un salaire négatif ou nul
    IF NEW.sal <= 0 THEN
        RAISE EXCEPTION 'Le salaire doit être positif, reçu : %', NEW.salaire;
    END IF;

    IF NEW.sal <= 450 THEN
        RAISE EXCEPTION 'Le salaire doit être superieur au SMIG , reçu : %', NEW.salaire;
    END IF;

    IF NEW.sal > 10000 THEN
        RAISE EXCEPTION 'Le salaire ne peut être superieur au plafond, reçu : %', NEW.salaire;
    END IF;



    RETURN NEW;   -- tout est OK, on laisse passer
END;
$$;

CREATE TRIGGER trg_valider_salaire
    BEFORE INSERT OR UPDATE
    ON storeuser.emp
    FOR EACH ROW
    EXECUTE FUNCTION valider_salaire();

CREATE OR REPLACE PROCEDURE storeuser.transaction_emp()
LANGUAGE plpgsql
AS $$
DECLARE
    n               INT;
    i               INT := 1;

    -- variables de calcul salaire
    v_sal_total     NUMERIC;
    v_sal_moyen     NUMERIC;
    v_sal_min       NUMERIC;
    v_sal_max       NUMERIC;

    v_emp           RECORD;

    -- prochain empno disponible
    v_next_empno    NUMERIC;

    v_erreur        BOOLEAN := FALSE;
BEGIN


    SELECT COUNT(*) INTO n FROM storeuser.emp;
    RAISE NOTICE 'Nombre d employés        : %', n;

    SELECT
        SUM(sal),
        ROUND(AVG(sal), 2),
        MIN(sal),
        MAX(sal)
    INTO v_sal_total, v_sal_moyen, v_sal_min, v_sal_max
    FROM storeuser.emp;

    RAISE NOTICE 'Masse salariale totale   : %', v_sal_total;
    RAISE NOTICE 'Salaire moyen            : %', v_sal_moyen;
    RAISE NOTICE 'Salaire minimum          : %', v_sal_min;
    RAISE NOTICE 'Salaire maximum          : %', v_sal_max;
    RAISE NOTICE 'MAX(empno) actuel        : %', (SELECT MAX(empno) FROM storeuser.emp);



    FOR v_emp IN
        SELECT
            e.empno,
            e.ename,
            e.job,
            e.sal,
            e.comm,
            d.dname,
            d.loc
        FROM storeuser.emp e
        JOIN storeuser.dept d ON d.deptno = e.deptno
        ORDER BY e.deptno, e.sal DESC
    LOOP
        RAISE NOTICE '  [%] % - % | sal=% | comm=% | dept=% (%)',
            v_emp.empno,
            v_emp.ename,
            v_emp.job,
            v_emp.sal,
            COALESCE(v_emp.comm::VARCHAR, 'N/A'),
            v_emp.dname,
            v_emp.loc;
    END LOOP;



    FOR v_emp IN
        SELECT empno, ename, job, sal
        FROM storeuser.emp
        ORDER BY empno
    LOOP
        DECLARE
            v_ancien  NUMERIC;
            v_nouveau NUMERIC;
            v_taux    NUMERIC;
        BEGIN
            v_ancien := v_emp.sal;

            -- Taux selon le job
            v_taux := CASE v_emp.job
                WHEN 'PRESIDENT' THEN 0.05   --  5%
                WHEN 'MANAGER'   THEN 0.10   -- 10%
                WHEN 'ANALYST'   THEN 0.12   -- 12%
                WHEN 'CLERK'     THEN 0.15   -- 15%
                WHEN 'SALESMAN'  THEN 0.08   --  8%
                ELSE                  0.05   --  5%
            END;

            v_nouveau := ROUND(v_emp.sal * (1 + v_taux), 2);

            UPDATE storeuser.emp
            SET sal = v_nouveau
            WHERE empno = v_emp.empno;

            RAISE NOTICE '  % [%] : % → % (+%% = +%)',
                v_emp.ename,
                v_emp.job,
                v_ancien,
                v_nouveau,
                v_taux * 100,
                ROUND(v_ancien * v_taux, 2);

        EXCEPTION
            WHEN check_violation THEN
                RAISE WARNING '  % : salaire invalide → ignoré', v_emp.ename;
                v_erreur := TRUE;
        END;

        i := i + 1;
    END LOOP;

    RAISE NOTICE '';
    RAISE NOTICE '── ÉTAPE 4 : Insertion de nouveaux employés ──';

    SELECT MAX(empno) INTO v_next_empno FROM storeuser.emp;

    DECLARE
        TYPE t_emp IS RECORD (
            ename   VARCHAR,
            job     VARCHAR,
            mgr     NUMERIC,
            sal     NUMERIC,
            comm    NUMERIC,
            deptno  NUMERIC
        );
    BEGIN
        -- Nouvel employé 1 → deptno 10 ACCOUNTING NEW YORK
        v_next_empno := v_next_empno + 100;
        INSERT INTO storeuser.emp (empno, ename, job, mgr, hiredate, sal, comm, deptno)
        VALUES (v_next_empno, 'MARTIN', 'CLERK', 7400, NOW(), 1100, NULL, 10);
        RAISE NOTICE '  ✓ Inséré empno=% MARTIN CLERK deptno=10', v_next_empno;

        -- Nouvel employé 2 → deptno 20 RESEARCH DALLAS
        v_next_empno := v_next_empno + 100;
        INSERT INTO storeuser.emp (empno, ename, job, mgr, hiredate, sal, comm, deptno)
        VALUES (v_next_empno, 'JOHNSON', 'ANALYST', 7400, NOW(), 2800, NULL, 20);
        RAISE NOTICE '  ✓ Inséré empno=% JOHNSON ANALYST deptno=20', v_next_empno;

        -- Nouvel employé 3 → deptno 30 SALES CHICAGO
        v_next_empno := v_next_empno + 100;
        INSERT INTO storeuser.emp (empno, ename, job, mgr, hiredate, sal, comm, deptno)
        VALUES (v_next_empno, 'ADAMS', 'SALESMAN', 7400, NOW(), 1600, 300, 30);
        RAISE NOTICE '  ✓ Inséré empno=% ADAMS SALESMAN deptno=30', v_next_empno;

    EXCEPTION
        WHEN foreign_key_violation THEN
            RAISE WARNING '  deptno inexistant → insertion annulée';
            v_erreur := TRUE;
        WHEN unique_violation THEN
            RAISE WARNING '  empno déjà existant → insertion annulée';
            v_erreur := TRUE;
        WHEN check_violation THEN
            RAISE WARNING '  sal <= 0 refusé par contrainte → insertion annulée';
            v_erreur := TRUE;
    END;

    RAISE NOTICE '';
    RAISE NOTICE '── ÉTAPE 5 : Attribution commissions SALESMAN ──';

    UPDATE storeuser.emp
    SET comm = ROUND(sal * 0.20, 2)
    WHERE job    = 'SALESMAN'
      AND comm IS NULL;

    GET DIAGNOSTICS n = ROW_COUNT;
    RAISE NOTICE '  % SALESMAN mis à jour avec comm = 20%% du sal', n;

    SELECT COUNT(*) INTO n FROM storeuser.emp;
    RAISE NOTICE 'Total employés après transaction : %', n;

    SELECT
        SUM(sal),
        ROUND(AVG(sal), 2)
    INTO v_sal_total, v_sal_moyen
    FROM storeuser.emp;

    RAISE NOTICE 'Nouvelle masse salariale         : %', v_sal_total;
    RAISE NOTICE 'Nouveau salaire moyen            : %', v_sal_moyen;
    RAISE NOTICE 'Nouveau MAX(empno)               : %', v_next_empno;

    RAISE NOTICE '';
    RAISE NOTICE '  Résumé par département :';
    FOR v_emp IN
        SELECT
            d.dname,
            d.loc,
            COUNT(e.empno)      AS nb,
            SUM(e.sal)          AS masse,
            ROUND(AVG(e.sal),2) AS moy
        FROM storeuser.dept d
        LEFT JOIN storeuser.emp e ON e.deptno = d.deptno
        GROUP BY d.dname, d.loc
        ORDER BY d.dname
    LOOP
        RAISE NOTICE '  % (%) → % emp | masse=% | moy=%',
            v_emp.dname,
            v_emp.loc,
            v_emp.nb,
            v_emp.masse,
            v_emp.moy;
    END LOOP;

    RAISE NOTICE '';
    IF v_erreur THEN
        RAISE NOTICE '⚠  Erreurs détectées → ROLLBACK';
        ROLLBACK;
    ELSE
        RAISE NOTICE '✓  Aucune erreur → COMMIT';
        COMMIT;
    END IF;

    RAISE NOTICE '';


END;
$$;

c.relname,

FROM pg_buffercache b
JOIN pg_class c
ON b.relfilenode = pg_relation_filenode(c.oid)
GROUP BY c.relname
ORDER BY buffers DESC;

               FROM pg_buffercache b JOIN pg_class c
               ON b.relfilenode = pg_relation_filenode(c.oid) AND
                  b.reldatabase IN (0, (SELECT oid FROM pg_database
                                        WHERE datname = current_database()))
               JOIN pg_namespace n ON n.oid = c.relnamespace
               GROUP BY n.nspname, c.relname
               ORDER BY 3 DESC
